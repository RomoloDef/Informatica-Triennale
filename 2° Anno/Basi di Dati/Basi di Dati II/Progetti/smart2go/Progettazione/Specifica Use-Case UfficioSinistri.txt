- calcolaClasseRischio(s: Socio): Intero >= 0

CREATE OR REPLACE FUNCTION calcolaClasseRischio(socio_id varchar)
RETURNS integer AS $$
DECLARE
    total_sinistri INTEGER := 0;
BEGIN
    -- Conta sinistri SENZA controparte
    SELECT COUNT(*)
    INTO total_sinistri
    FROM Sinistri s
    JOIN SinistriSenzaControparte ssc ON s.id = ssc.sinistro
    JOIN Noleggio n ON s.noleggio = n.id
    JOIN effettua e ON n.id = e.noleggio
    WHERE e.socio = socio_id
      AND s.quando >= NOW() - INTERVAL '3 years';

    -- Aggiunge sinistri CON controparte
    SELECT total_sinistri + COUNT(*)
    INTO total_sinistri
    FROM Sinistri s
    JOIN SinistriConControparte scp ON s.id = scp.sinistro
    JOIN Noleggio n ON s.noleggio = n.id
    JOIN effettua e ON n.id = e.noleggio
    WHERE e.socio = socio_id
      AND s.quando >= NOW() - INTERVAL '3 years';

    -- Calcolo classe di rischio
    IF total_sinistri = 0 THEN
        RETURN 0;
    ELSIF total_sinistri < 10 THEN
        RETURN total_sinistri;
    ELSE
        RETURN 10;
    END IF;
END;
$$ LANGUAGE plpgsql;
