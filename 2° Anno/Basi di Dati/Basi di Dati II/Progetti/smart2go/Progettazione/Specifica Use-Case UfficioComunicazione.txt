        )

        FROM Auto a
        JOIN Noleggio n ON a.targa = n.auto
        JOIN effettua e ON n.id = e.noleggio
        WHERE e.socio = :socio_id
        AND (SELECT COUNT(*) FROM categorie_noleggiate) = 1;
