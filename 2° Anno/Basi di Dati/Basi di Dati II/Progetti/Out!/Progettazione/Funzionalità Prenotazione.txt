- posti_disponibili(s:Sala): Intero 

pre:

post:

    algoritmo:

    WITH prenotati AS (
    SELECT t.sala, COUNT(*) AS posti_prenotati
    FROM pren_post pr, trovatosi t
    WHERE pr.fila = t.fila
      AND pr.colonna = t.colonna
    GROUP BY t.sala
    )

    SELECT s.id,
        s.capienza - (
            SELECT COUNT(*)
            FROM pren_post pr, trovatosi t
            WHERE pr.fila = t.fila
                AND pr.colonna = t.colonna
                AND t.sala = s.id
        ) AS posti_disponibili
    FROM Sala s;
