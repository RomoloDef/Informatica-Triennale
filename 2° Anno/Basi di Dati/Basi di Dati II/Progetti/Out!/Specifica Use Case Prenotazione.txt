- posti_disponibili(s: Sala): Intero 

pre:

    EXIST trovatosi(p, s)

post:

    S = { s | trovatosi(p, s)}

    P = { p | Prenotazione(pr) AND preno_post(pr, p)}

    result = s.capienza - SUM(p.posti)
