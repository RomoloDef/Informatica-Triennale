- operazione ausiliaria numGiorni()

CREATE OR REPLACE FUNCTION numGiorni(mese TEXT)
RETURNS INTEGER
AS $$
BEGIN
    CASE 
        WHEN mese IN ('Aprile', 'Giugno', 'Settembre', 'Novembre') THEN
            RETURN 30;
        WHEN mese = 'Febbraio' THEN
            RETURN 28;
        ELSE
            RETURN 31;
    END CASE;
END;
$$ LANGUAGE plpgsql;
