- operazione ausiliaria: contiene()

CREATE OR REPLACE FUNCTION contiene(ora TIME, inizio TIME, fine TIME)
RETURNS BOOLEAN
AS $$
BEGIN
    RETURN ora BETWEEN inizio AND fine;
END;
$$ LANGUAGE plpgsql;
